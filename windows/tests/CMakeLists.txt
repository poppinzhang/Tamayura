cmake_minimum_required(VERSION 3.14)
project(tamayura_native_tests LANGUAGES CXX)
enable_testing()

add_executable(discovery_core_test discovery_core_test.cpp ../runner/discovery_core.cpp)
target_compile_features(discovery_core_test PRIVATE cxx_std_17)
target_include_directories(discovery_core_test PRIVATE ../runner)
add_test(NAME discovery_core_test COMMAND discovery_core_test)

if(WIN32)
  add_executable(windows_update_old_fixture update_fixture.cpp update_fixture.rc)
  add_executable(windows_update_new_fixture update_fixture.cpp update_fixture.rc)
  target_compile_definitions(windows_update_old_fixture PRIVATE UPDATE_PATCH=3 UPDATE_BUILD=4)
  target_compile_definitions(windows_update_new_fixture PRIVATE UPDATE_PATCH=4 UPDATE_BUILD=5)
  add_test(NAME windows_update_test COMMAND powershell.exe -NoLogo -NoProfile
    -NonInteractive -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/update_test.ps1"
    -OldExecutable "$<TARGET_FILE:windows_update_old_fixture>"
    -NewExecutable "$<TARGET_FILE:windows_update_new_fixture>")
  set_tests_properties(windows_update_test PROPERTIES TIMEOUT 120)
  add_executable(windows_desktop_tray_test desktop_tray_test.cpp ../runner/desktop_tray.cpp
    ../runner/app_text.cpp)
  add_executable(windows_app_text_test app_text_test.cpp ../runner/app_text.cpp)
  target_compile_features(windows_app_text_test PRIVATE cxx_std_17)
  target_include_directories(windows_app_text_test PRIVATE ../runner)
  target_compile_definitions(windows_app_text_test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN
    UNICODE _UNICODE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  target_compile_features(windows_desktop_tray_test PRIVATE cxx_std_17)
  target_include_directories(windows_desktop_tray_test PRIVATE ../runner)
  target_compile_definitions(windows_desktop_tray_test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN
    UNICODE _UNICODE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  target_link_libraries(windows_desktop_tray_test PRIVATE shell32 user32)
  add_executable(windows_storage_test storage_test.cpp ../runner/storage_core.cpp)
  add_executable(windows_preview_test preview_test.cpp ../runner/storage_core.cpp)
  add_executable(windows_thumbnail_cache_test thumbnail_cache_test.cpp ../runner/storage_core.cpp)
  foreach(storage_target windows_storage_test windows_preview_test windows_thumbnail_cache_test)
    target_compile_features(${storage_target} PRIVATE cxx_std_17)
    target_include_directories(${storage_target} PRIVATE ../runner)
    target_compile_definitions(${storage_target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN
      UNICODE _UNICODE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
    target_link_libraries(${storage_target} PRIVATE shell32 ole32 windowscodecs bcrypt gdi32 uuid)
    if(MSVC)
      target_compile_options(${storage_target} PRIVATE /W4 /WX /EHsc /utf-8)
    endif()
    add_test(NAME ${storage_target} COMMAND ${storage_target})
  endforeach()
  if(MSVC)
    target_compile_options(windows_desktop_tray_test PRIVATE /W4 /WX /EHsc /utf-8)
    target_compile_options(windows_app_text_test PRIVATE /W4 /WX /EHsc /utf-8)
    target_compile_options(discovery_core_test PRIVATE /W4 /WX /EHsc /utf-8)
  endif()
  add_test(NAME windows_desktop_tray_test COMMAND windows_desktop_tray_test)
  add_test(NAME windows_app_text_test COMMAND windows_app_text_test)
endif()
